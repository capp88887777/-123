function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); var list = []; // 첫 번째 행은 헤더이므로 i=1부터 시작 for (var i = 1; i < rows.length; i++) { var r = rows[i]; if (r[1] && String(r[1]).trim() !== "") { // 이름이 있는 경우만 수집 list.push({ id: String(r[0] || "member_" + i), name: String(r[1] || ""), subs: String(r[2] || ""), king: String(r[3] || "보유 없음"), is2nd: r[4] === true || r[4] === "true" || r[4] === "TRUE", isAwaken: r[5] === true || r[5] === "true" || r[5] === "TRUE", duty: String(r[6] || ""), note: String(r[7] || ""), position: String(r[8] || "상단명단") }); } } return ContentService.createTextOutput(JSON.stringify(list)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(10000); // 동시 요청 충돌 방지 락 var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); // 시트 초기화 후 헤더 재작성 sheet.clear(); sheet.appendRow(["ID", "본캐 닉네임", "부캐릭터", "사천왕", "2차 완료", "각성 완료", "직책", "비고", "직책분류"]); var rowsToAppend = []; // 중복 방지를 위해 고유 ID 기준으로 데이터 정제 (Map 이용) var uniqueMap = new Map(); data.forEach(function(m) { if (m && m.name && String(m.name).trim() !== "") { var id = String(m.id || ("member_" + Date.now() + "_" + Math.random())); uniqueMap.set(id, [ id, String(m.name || ""), String(m.subs || ""), String(m.king || "보유 없음"), !!m.is2nd, !!m.isAwaken, String(m.duty || ""), String(m.note || ""), String(m.position || "상단명단") ]); } }); uniqueMap.forEach(function(row) { rowsToAppend.push(row); }); if (rowsToAppend.length > 0) { sheet.getRange(2, 1, rowsToAppend.length, rowsToAppend[0].length).setValues(rowsToAppend); } return ContentService.createTextOutput(JSON.stringify({status: "success", count: rowsToAppend.length})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
